    const ownSettings = settingsFor(zoneId);

      !ownSettings.mirror ||
      !ownSettings.syncTarget ||

      return {
        ...presetApi.clone(ownSettings),
        colors: presetApi.normalizeColors(ownSettings)

      ownSettings.syncTarget,
      visited

      sourcePreset.directional &&
      ownSettings.mirrorDirection === "reverse"

      direction,
      syncTarget: ownSettings.syncTarget,

        ...presetApi.clone(preservedSettings),
        syncTarget: "",
        mirror: false,
        mirrorDirection: "same"

      return;

    zoneSettings.set(zoneId, {
      ...presetApi.clone(ownSettings),
      syncTarget: "",
      mirror: false,
      mirrorDirection: "same"
    });
  }

  function zoneSeed(zoneId) {
    let hash = 2166136261;

    for (const character of zoneId) {
      hash = Math.imul(hash ^ character.charCodeAt(0), 16777619);
    }

    return (hash >>> 0) % 997;
  }

  function mixColors(colorA, colorB, amount) {
    return colorA.map((value, index) => (
      value + ((colorB[index] - value) * amount)
    ));
  }

  function colorForPosition(settings, position) {
    const colors = presetApi
      .normalizeColors(settings)
      .map(presetApi.hexToRgb);
    const normalizedPosition = ((position % 1) + 1) % 1;
    const scaledPosition = normalizedPosition * (colors.length - 1);
    const startIndex = Math.floor(scaledPosition);
    const amount = scaledPosition - startIndex;
    const endIndex = Math.min(startIndex + 1, colors.length - 1);

    return mixColors(
      colors[startIndex],
      colors[endIndex],
      amount
    );
  }

  // BEGIN OBJECT: Approved geometry validation.
  function validateGeometry(payload) {
    if (!payload || payload.locked !== true) {
      throw new Error("Home geometry is not marked as locked.");
    }

    if (
      payload.coordinateSystem?.width !== 1448 ||
      payload.coordinateSystem?.height !== 1086
    ) {
      throw new Error("Home geometry coordinate system is invalid.");
    }

    for (const sceneName of ["front", "rear"]) {
      const scene = (payload.scenes || payload.views)?.[sceneName];

      if (!scene || !Array.isArray(scene.zones) || !scene.zones.length) {
        throw new Error(`Home ${sceneName} geometry is missing.`);
      }

      if (scene.width !== 1448 || scene.height !== 1086) {
        throw new Error(`Home ${sceneName} image dimensions are invalid.`);
      }
    }

    return payload;
  }

  // BEGIN API CALL: Load locked Home SIM geometry.
  async function loadGeometry() {
    const geometryUrl = new URL(GEOMETRY_PATH, siteRoot);
    geometryUrl.searchParams.set("v", "20260805-production-home-v1");

    const response = await fetch(geometryUrl, {
      cache: "no-store",
      credentials: "same-origin"
    });

    if (!response.ok) {
      throw new Error(`Geometry request failed with ${response.status}.`);
    }

    return validateGeometry(await response.json());
  }

  async function loadSceneImage(sceneName) {
    if (sceneImageCache.has(sceneName)) {
      return sceneImageCache.get(sceneName);
    }

    const scene = scenes[sceneName];
    const imagePath = scene.image || APPROVED_IMAGE_PATHS[sceneName];
    const assetUrl = new URL(imagePath, siteRoot);
    assetUrl.searchParams.set("v", "20260805-approved-home-image-v1");

    const response = await fetch(assetUrl, {
      cache: "force-cache",
      credentials: "same-origin"
    });

    if (!response.ok) {
      throw new Error(`Approved ${sceneName} image failed with ${response.status}.`);
    }

    const base64Image = (await response.text()).trim();

    if (!base64Image.startsWith("UklG")) {
      throw new Error(`Approved ${sceneName} image data is invalid.`);
    }

    const imageSource = `data:image/webp;base64,${base64Image}`;

    sceneImageCache.set(sceneName, imageSource);

    return imageSource;
  }

  // BEGIN OBJECT: Zone path and LED particle generation.
  function pathData(run) {
    const points = run.points || [];

    if (!points.length) {
      return "";
    }

    let data = `M ${points[0][0]} ${points[0][1]}`;

    for (let index = 1; index < points.length; index += 1) {
      data += ` L ${points[index][0]} ${points[index][1]}`;
    }

    if (run.kind === "closed") {
      data += " Z";
    }

    return data;
  }

  function samplePolyline(points, spacing, closed = false) {
    if (!Array.isArray(points) || points.length < 2) {
      return points || [];
    }

    const sourcePoints = closed
      ? [...points, points[0]]
      : [...points];
    const segments = [];
    let totalLength = 0;

    for (let index = 1; index < sourcePoints.length; index += 1) {
      const start = sourcePoints[index - 1];
      const end = sourcePoints[index];
      const length = Math.hypot(
        end[0] - start[0],
        end[1] - start[1]
      );

      segments.push({
        start,
        end,
        length,
        distanceStart: totalLength
      });
      totalLength += length;
    }

    if (!totalLength) {
      return [sourcePoints[0]];
    }

    const count = Math.max(
      2,
      Math.floor(totalLength / Math.max(1, spacing)) + 1
    );
    const sampledPoints = [];

    for (let index = 0; index < count; index += 1) {
      const distance = closed
        ? (index / count) * totalLength
        : (index / (count - 1)) * totalLength;
      const segment = segments.find((item) => (
        distance <= item.distanceStart + item.length
      )) || segments[segments.length - 1];
      const segmentProgress = segment.length
        ? (distance - segment.distanceStart) / segment.length
        : 0;

      sampledPoints.push([
        segment.start[0] + (
          (segment.end[0] - segment.start[0]) * segmentProgress
        ),
        segment.start[1] + (
          (segment.end[1] - segment.start[1]) * segmentProgress
        )
      ]);
    }

    return sampledPoints;
  }
